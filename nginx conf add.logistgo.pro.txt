server {
    server_name add.logistgo.pro ;
    listen 10.1.1.215:443 ssl;
    listen 10.1.1.215:443 quic;
        

    ssl_certificate "/var/www/httpd-cert/add.logistgo.pro_2026-02-08-16-39_17.crt";
    ssl_certificate_key "/var/www/httpd-cert/add.logistgo.pro_2026-02-08-16-39_17.key";
    add_header Strict-Transport-Security "max-age=31536000" always;


    http2 on;

    ssl_protocols TLSv1.1 TLSv1.2 TLSv1.3;
    add_header Alt-Svc 'h3=":443"; ma=86400';

    charset utf-8;

    gzip on;
    gzip_min_length 1024;
    gzip_proxied expired no-cache no-store private auth;
    gzip_types text/css image/x-ico application/pdf image/jpeg image/png image/gif application/javascript application/x-javascript application/x-pointplus;
    gzip_comp_level 4;
    
    set $root_path /var/www/logistgo/data/www/add.logistgo.pro;
    root $root_path;
    disable_symlinks if_not_owner from=$root_path;

# API router    
    location ^~ /frontend/api/ {
        rewrite ^/frontend/api/(.*)$ /frontend/api/index.php?r=/$1 break;
        proxy_pass http://127.0.0.1:81;
        proxy_redirect http://127.0.0.1:81/ /;
        include /etc/nginx/proxy_params;
    }

    location / {
        proxy_pass http://127.0.0.1:81;
        proxy_redirect http://127.0.0.1:81/ /;
        include /etc/nginx/proxy_params;
    }


     location ~* ^.+\.(jpg|jpeg|gif|png|svg|js|css|mp3|ogg|mpeg|avi|zip|gz|bz2|rar|swf|ico|7z|doc|docx|map|ogg|otf|pdf|ttf|tif|txt|wav|webp|woff|woff2|xls|xlsx|xml)$ {
        try_files $uri $uri/ @fallback;
        expires 360d;

    }

    location @fallback {
        proxy_pass http://127.0.0.1:81;
        proxy_redirect http://127.0.0.1:81/ /;
        include /etc/nginx/proxy_params;
    }

    include "/etc/nginx/fastpanel2-sites/logistgo/add.logistgo.pro.includes";
    include /etc/nginx/fastpanel2-includes/*.conf;

    error_log /var/www/logistgo/data/logs/add.logistgo.pro-frontend.error.log;
    access_log /var/www/logistgo/data/logs/add.logistgo.pro-frontend.access.log;
}
